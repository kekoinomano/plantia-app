cmake_minimum_required(VERSION 3.18)
project(plantia_mood_render LANGUAGES C CXX)
set(SFIZZ_JACK OFF CACHE BOOL "" FORCE)
set(SFIZZ_RENDER OFF CACHE BOOL "" FORCE)
set(SFIZZ_SHARED OFF CACHE BOOL "" FORCE)
set(SFIZZ_GIT_SUBMODULE_CHECK OFF CACHE BOOL "" FORCE)
set(SFIZZ_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../modules/plantia-pcm/third_party/sfizz-1.2.3 sfizz)
add_executable(plantia-mood-render main.cpp)
target_include_directories(plantia-mood-render PRIVATE ../../modules/plantia-pcm/cpp)
target_link_libraries(plantia-mood-render PRIVATE sfizz::sfizz)
target_compile_features(plantia-mood-render PRIVATE cxx_std_17)
target_compile_options(plantia-mood-render PRIVATE -O3 -ffp-contract=off)
